package com.example.demo.service;

import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.chat.memory.ChatMemory;
import org.springframework.ai.chat.memory.MessageWindowChatMemory;
import org.springframework.ai.mcp.SyncMcpToolCallbackProvider;
import org.springframework.ai.chat.client.advisor.MessageChatMemoryAdvisor;
import org.springframework.stereotype.Service;

import com.example.demo.enums.AiAgent;


@Service
public class AiAgentService {

    private final ChatClient generalAgent;
    private final ChatClient codingAgent;
    private final ChatClient ragAgent;
    private final ChatClient mcpAgent;

    private final ChatMemory chatMemory;

   public AiAgentService(ChatClient.Builder builder, SyncMcpToolCallbackProvider mcpTools) {

        this.chatMemory = MessageWindowChatMemory
                .builder()
                .maxMessages(20)
                .build();

        this.generalAgent = createAgent(builder, mcpTools);

        this.codingAgent = createAgent(builder, mcpTools);

        this.ragAgent = createAgent(builder, mcpTools);

        this.mcpAgent = createAgent(builder, mcpTools);
    }

    private ChatClient createAgent(
            ChatClient.Builder builder,
            SyncMcpToolCallbackProvider mcpTools) {

        return builder.clone()
                .defaultToolCallbacks(mcpTools.getToolCallbacks())
                .defaultAdvisors(MessageChatMemoryAdvisor.builder(chatMemory).build())
                .build();
    }

}
